class CSAM(nn.Module):
    def __init__(self, c, r=16):
        super().__init__()
        self.mlp = nn.Sequential(nn.Linear(c, c // r), nn.ReLU(),
                                 nn.Linear(c // r, c))
        self.conv = nn.Conv2d(2, 1, 7, padding=3)

    def forward(self, x):
        ca = torch.sigmoid(self.mlp(x.mean((2, 3)))).view(*x.shape[:2], 1, 1)
        x = x * ca                                  # 通道注意力
        sa = torch.sigmoid(self.conv(torch.cat(
            [x.mean(1, keepdim=True), x.amax(1, keepdim=True)], 1)))
        return x * sa                              # 空间注意力
