import pandas as pd
import matplotlib.pyplot as plt

plt.rcParams.update({"font.family": "Arial", "font.size": 8, "pdf.fonttype": 42})

df = pd.read_csv("results/acc.csv")          # 数字全部来自你的数据文件
fig, ax = plt.subplots(figsize=(8.5 / 2.54, 6.5 / 2.54), dpi=300)  # 单栏 8.5 cm

styles = {"Baseline": ("#8a8f9c", "--"),
          "+ Module A": ("#e08a2c", "-"),
          "Ours": ("#2f6fed", "-")}
for name, (color, ls) in styles.items():
    ax.plot(df["epoch"], df[name], color=color, linestyle=ls,
            linewidth=1.2, label=name)

ax.set_xlabel("Epoch")
ax.set_ylabel("Validation accuracy (%)")
ax.legend(frameon=False)
fig.tight_layout()
fig.savefig("figures/acc.pdf")                 # 矢量图，放大不糊
fig.savefig("figures/acc.png")
